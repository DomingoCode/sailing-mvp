  <h1>⛵ Sailing MVP</h1>


  <button onclick="loadData()">Load trips</button>


  <div id="list"></div>


  <script>

  const API = "https://sail-api-nq2m.onrender.com";


  async function loadData() {

    const res = await fetch(API + "/listings");

    const data = await res.json();


    const list = document.getElementById("list");

    list.innerHTML = "";


    data.forEach(item => {

      const div = document.createElement("div");

      div.className = "card";

      div.innerHTML = `

        <h3>${item.title}</h3>

        <p>${item.location}</p>

        <p>${item.dates}</p>

        <p>${item.price}</p>

        <a href="${item.url}" target="_blank">source</a>

      `;

      list.appendChild(div);

    });

  }

</script>
